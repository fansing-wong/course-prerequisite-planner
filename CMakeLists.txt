cmake_minimum_required(VERSION 3.21)
project(course_plan LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_library(course_plan_core course_plan_core.cpp)
target_include_directories(course_plan_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})

add_executable(course_plan course_plan.cpp)
target_link_libraries(course_plan PRIVATE course_plan_core)

include(CTest)
if(BUILD_TESTING)
    add_executable(course_plan_core_tests tests/course_plan_core_tests.cpp)
    target_link_libraries(course_plan_core_tests PRIVATE course_plan_core)
    add_test(NAME course_plan_core_tests COMMAND course_plan_core_tests)
endif()

function(enable_project_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /utf-8)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -pedantic)
        if(WIN32 AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
            target_compile_options(${target} PRIVATE -finput-charset=UTF-8 -fexec-charset=UTF-8)
        endif()
    endif()
endfunction()

enable_project_warnings(course_plan_core)
enable_project_warnings(course_plan)
if(BUILD_TESTING)
    enable_project_warnings(course_plan_core_tests)
endif()
